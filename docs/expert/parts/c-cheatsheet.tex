\section*{Commands}

\begin{center}
\begin{tabularx}{\linewidth}{lX}
\toprule
\code{nodwyn} & a session in this folder (\code{-{}-repo DIR}, \code{-{}-resume ID}) \\
\code{nodwyn app} & the console in a window; runs a daemon if none is running \\
\code{nodwyn ui [-{}-listen ADDR]} & the console in a browser; \code{-{}-listen} for other devices \\
\code{nodwyn tui} & the full-screen terminal client \\
\code{nodwyn daemon} & the session server \\
\code{nodwyn run -{}-task F -{}-max-turns N -{}-budget-wall D} & a supervised run, \code{-{}-unattended} for nobody to answer it, \code{-{}-yolo} or \code{-{}-judge} for its confirms; \code{status}, \code{stop}, \code{resume} \\
\code{nodwyn sessions}, \code{replay ID [-{}-window SEQ]} & list; transcript; exact request \\
\code{nodwyn summary \ldots} & add, list, drop reviewed summaries \\
\code{nodwyn versions FILE}, \code{undo FILE} & kept versions outside git \\
\code{nodwyn devices [revoke NAME]} & paired remote browsers \\
\code{nodwyn keychain init}, \code{keychain passphrase} & create the keychain (\code{-{}-no-passphrase}: none); set, change or \code{-{}-remove} its passphrase \\
\code{nodwyn mcp [list]}, \code{mcp pull NAME} & MCP servers; download a contained server's image by digest \\
\code{nodwyn credential \ldots} & \code{list} the keychain's credentials, \code{revoke ID} one; \code{add -{}-oauth2 P} and \code{login ID} authorize OAuth2 in the browser \\
\code{nodwyn repl}, \code{eval -e EXPR}, \code{eval FILE} & Scheme alone \\
\code{nodwyn paths}, \code{version} & where things live; which build \\
\bottomrule
\end{tabularx}
\end{center}

\section*{Session flags}

\code{-{}-yolo} (ordinary confirms yes), \code{-{}-no-confirm} (every confirm
no), \code{-{}-max-tool-calls N}, \code{-{}-budget-wall 2h},
\code{-{}-budget-tokens N}, \code{-{}-budget-actions N}, \code{-{}-model},
\code{-{}-url}, \code{-{}-show-thinking}, \code{-{}-no-keychain}.

\section*{The init.scm skeleton}

\begin{scm}
;; .nodwyn/init.scm
(load ".nodwyn/tools.scm")                   ; register-tool ...

(register-check "hands-off-generated" "deny" ; hard rules
  (lambda (prim args)
    (and (member prim '("fs-write" "fs-edit"))
         (pair? args)
         (string-prefix? "gen/" (format "~a" (car args))))))

(define base-system-prompt system-prompt)    ; project facts
(define (system-prompt repo)
  (string-append (base-system-prompt repo) "\n\n"
    """
    Tests: ./run-tests. Never edit gen/.
    """))

(define test-command-words                   ; teach the evidence loop
  (cons "./run-tests" test-command-words))

(define my-base-run-calls run-calls)         ; a strategy layer
(define (run-calls calls)
  (my-base-run-calls calls))                 ; one result per call, in order
\end{scm}
